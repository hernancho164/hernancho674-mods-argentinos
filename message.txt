        function reproducirClick() {
            const clickAudio = document.getElementById('click-audio');

        function cambiarSeccion(idSeccion, event) {
            reproducirClick();

            const secciones = document.querySelectorAll('.section-content');
            secciones.forEach(sec => sec.classList.remove('active'));

            const tabs = document.querySelectorAll('.nav-tabs .tab');
            tabs.forEach(t => t.classList.remove('active'));

            document.getElementById(idSeccion).classList.add('active');
            
            if(['cs16', 'cssource', 'csother'].includes(idSeccion)) {
                event.currentTarget.closest('.dropdown').querySelector('.tab').classList.add('active');
            } else if(['tf2', 'tf2classified'].includes(idSeccion)) {
                event.currentTarget.closest('.dropdown').querySelector('.tab').classList.add('active');
            } else if(['hl1', 'hl1opposing'].includes(idSeccion)) {
                event.currentTarget.closest('.dropdown').querySelector('.tab').classList.add('active');
            } else if(['dodsource', 'ricochet', 'l4d2'].includes(idSeccion)) {
                event.currentTarget.closest('.dropdown').querySelector('.tab').classList.add('active');
            } else {
                event.currentTarget.classList.add('active');
            }
        }

        let proyectoActualIndex = 0;

        function cambiarProyecto(direccion) {
            reproducirClick();
            proyectoActualIndex += direccion;
            
            if (proyectoActualIndex < 0) {
                proyectoActualIndex = proyectosData.length - 1;
            } else if (proyectoActualIndex >= proyectosData.length) {
                proyectoActualIndex = 0;
            }

            const p = proyectosData[proyectoActualIndex];
            const imgEl = document.getElementById('gta-img');
            const badgeEl = document.getElementById('gta-badge');
            const tituloEl = document.getElementById('gta-titulo');
            const descEl = document.getElementById('gta-descripcion');

            imgEl.style.opacity = '0';
            setTimeout(() => {
                imgEl.src = p.imagen;
                tituloEl.textContent = p.titulo;
                
                // Mantenemos los saltos de línea con <br> para que quede prolijo en HTML
                descEl.innerHTML = p.descripcion.replace(/\n/g, '<br>');
                
                badgeEl.textContent = p.estado;
                badgeEl.className = 'gta-estado-badge ' + p.claseEstado;
                imgEl.style.opacity = '0.55';
            }, 200);
        }

        let clicksSerj = 0;
        let estaAusente = false;
        let serjCooldown = false;

        function interactuarSerj(e) {
            if (estaAusente || serjCooldown) return;

            const widget = document.getElementById('serj-widget');
            const serjImg = document.getElementById('serj-img');
            const bubble = document.getElementById('serj-bubble');
            clicksSerj++;

            if (clicksSerj >= 5) {
                estaAusente = true;
                serjCooldown = true;
                clicksSerj = 0;

                bubble.style.display = 'none';
                serjImg.classList.remove('serj-shaking');
                serjImg.classList.remove('serj-walking');

                serjImg.src = 'serj2.png';
                serjImg.classList.add('serj-shaking');
                bubble.style.display = 'block';

                const grito = document.getElementById('grito-audio');
                grito.currentTime = 0;
                grito.play().catch(err => console.log("Audio bloqueado:", err));

                setTimeout(() => {
                    bubble.style.display = 'none';
                    serjImg.classList.remove('serj-shaking');
                    serjImg.classList.add('serj-walking');
                }, 2500);

                setTimeout(() => {
                    serjImg.style.display = 'none';
                    widget.style.display = 'none';
                }, 3500);

                setTimeout(() => {
                    serjImg.src = 'serj.png';
                    serjImg.classList.remove('serj-walking');
                    serjImg.style.display = 'block';
                    widget.style.display = 'block';
                    widget.style.pointerEvents = 'auto';
                    serjImg.style.opacity = '1';
                    serjImg.style.transform = 'scale(1) rotate(0deg)';
                    estaAusente = false;
                    serjCooldown = false;
                }, 48500);

            } else {
                if (serjImg.style.opacity === '0') return;

                serjCooldown = true;
                setTimeout(() => { serjCooldown = false; }, 300);

                widget.style.pointerEvents = 'none';

                const audio = document.getElementById('wakeup-audio');
                audio.currentTime = 0;
                audio.play().catch(err => console.log("Audio bloqueado:", err));

                const cantidadPistachos = 12;
                for (let i = 0; i < cantidadPistachos; i++) {
                    const pistacho = document.createElement('img');
                    pistacho.src = 'pistacho.png';
                    pistacho.className = 'pistacho-efecto';

                    const angulo = -Math.PI / 4 - Math.random() * (Math.PI / 2);
                    const distancia = 150 + Math.random() * 250;
                    const tx = Math.cos(angulo) * distancia;
                    const ty = Math.sin(angulo) * distancia;
                    const rotacion = (Math.random() - 0.5) * 720;

                    pistacho.style.setProperty('--tx', `${tx}px`);
                    pistacho.style.setProperty('--ty', `${ty}px`);
                    pistacho.style.setProperty('--rot', `${rotacion}deg`);

                    document.body.appendChild(pistacho);

                    setTimeout(() => {
                        pistacho.remove();
                    }, 900);
                }

                serjImg.style.transition = 'opacity 0.3s ease, transform 0.3s ease';
                serjImg.style.opacity = '0';
                serjImg.style.transform = 'scale(0.2) rotate(180deg)';

                setTimeout(() => {
                    serjImg.style.transition = 'opacity 0.5s ease, transform 0.5s ease';
                    serjImg.style.opacity = '1';
                    serjImg.style.transform = 'scale(1) rotate(0deg)';
                    
                    widget.style.pointerEvents = 'auto';
                }, 3000);
            }
        }

        let typedBuffer = '';
        const secretCode = 'edm';
        let edmTriggered = false;

        let typedBufferBaneado = '';
        let baneadoTriggered = false;

        window.addEventListener('keydown', (e) => {
            if (!sitioEntrado) return;
            if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;

            const key = e.key.toLowerCase();

            // --- EASTER EGG EDM ---
            if (!edmTriggered) {
                typedBuffer += key;
                if (typedBuffer.length > 10) {
                    typedBuffer = typedBuffer.slice(-10);
                }
                if (typedBuffer.includes(secretCode)) {
                    typedBuffer = '';
                    triggerEDMEasterEgg();
                    return;
                }
            }

            // --- EASTER EGG BANEADO ("sixseven" o "67") ---
            if (!baneadoTriggered) {
                typedBufferBaneado += key;
                if (typedBufferBaneado.length > 15) {
                    typedBufferBaneado = typedBufferBaneado.slice(-15);
                }
                if (typedBufferBaneado.includes('sixseven') || typedBufferBaneado.includes('67')) {
                    typedBufferBaneado = '';
                    triggerBaneadoEasterEgg();
                }
            }
        });

        function triggerEDMEasterEgg() {
            edmTriggered = true;
            window.scrollTo({ top: 0, behavior: 'instant' });
            document.body.classList.add('bloqueado-total');

            const staticOv = document.getElementById('static-overlay');
            const jumpscare = document.getElementById('jumpscare-container');
            const blackout = document.getElementById('blackout-screen');

            const audioDross = document.getElementById('audio-dross');
            const audioCabronazo = document.getElementById('audio-cabronazo');
            const audioRisa = document.getElementById('audio-risa');

            audioDross.currentTime = 0;
            audioDross.play().catch(err => console.log("Audio Dross bloqueado:", err));
            document.body.classList.add('modo-alerta');

            setTimeout(() => {
                staticOv.style.opacity = '0.4';
                jumpscare.style.opacity = '1';
                audioCabronazo.currentTime = 0;
                audioCabronazo.play().catch(err => console.log("Audio Cabronazo bloqueado:", err));
            }, 3000);

            setTimeout(() => {
                jumpscare.style.opacity = '0';
                staticOv.style.opacity = '0';
                blackout.style.opacity = '1';
            }, 4600);

            setTimeout(() => {
                document.body.classList.remove('modo-alerta');
                blackout.style.opacity = '0';
                audioRisa.currentTime = 0;
                audioRisa.play().catch(err => console.log("Audio risa bloqueado:", err));
                
                document.body.classList.remove('bloqueado-total');
                edmTriggered = false;
            }, 6500);
        }

        function triggerBaneadoEasterEgg() {
            baneadoTriggered = true;
            document.body.classList.add('bloqueado-total');

            const audioBaneado = document.getElementById('audio-baneado');
            audioBaneado.currentTime = 0;
            audioBaneado.play().catch(err => console.log("Audio baneado bloqueado:", err));

            setTimeout(() => {
                window.close();
                window.location.href = "about:blank";
            }, 2000);
        }
